26inline bool decompressLZF(const std::vector<unsigned char>& input,
27 std::vector<unsigned char>& output) {
28 std::size_t ip = 0;
29 std::size_t op = 0;
30
31 while (ip < input.size()) {
32 unsigned int ctrl = input[ip++];
33 if (ctrl < (1u << 5)) {
34 ctrl += 1;
35 if (op + ctrl > output.size() || ip + ctrl > input.size()) {
36 return false;
37 }
38 std::memcpy(output.data() + op, input.data() + ip, ctrl);
39 op += ctrl;
40 ip += ctrl;
41 } else {
42 unsigned int len = ctrl >> 5;
43 if (ip >= input.size()) {
44 return false;
45 }
46 std::size_t ref = op - ((ctrl & 0x1f) << 8) - 1;
47 if (len == 7) {
48 len += input[ip++];
49 if (ip >= input.size()) {
50 return false;
51 }
52 }
53 ref -= input[ip++];
54
55 if (ref >= op || op + len + 2 > output.size()) {
56 return false;
57 }
58
59 output[op++] = output[ref++];
60 output[op++] = output[ref++];
61 while (len--) {
62 output[op++] = output[ref++];
63 }
64 }
65 }
66
67 return op == output.size();
68}

70inline bool parsePCDHeader(std::istream& stream, PCDHeader& header) {
71 std::string line;
72 while (std::getline(stream, line)) {
73 if (line.empty() || line[0] == '#') {
74 continue;
75 }
76
77 std::stringstream ss(line);
78 std::string tag;
79 ss >> tag;
80 if (tag == "FIELDS") {
81 header.fields.clear();
82 std::string field;
83 while (ss >> field) {
84 header.fields.push_back(field);
85 }
86 } else if (tag == "SIZE") {
87 header.sizes.clear();
88 int value = 0;
89 while (ss >> value) {
90 header.sizes.push_back(value);
91 }
92 } else if (tag == "TYPE") {
93 header.types.clear();
94 char value = '\0';
95 while (ss >> value) {
96 header.types.push_back(value);
97 }
98 } else if (tag == "COUNT") {
99 header.counts.clear();
100 int value = 0;
101 while (ss >> value) {
102 header.counts.push_back(value);
103 }
104 } else if (tag == "POINTS") {
105 ss >> header.points;
106 } else if (tag == "WIDTH" && header.points == 0) {
107 ss >> header.points;
108 } else if (tag == "DATA") {
109 ss >> header.data_type;
110 break;
111 }
112 }
113
114 if (header.fields.empty() || header.sizes.size() != header.fields.size() ||
115 header.types.size() != header.fields.size()) {
116 return false;
117 }
118 if (header.counts.empty()) {
119 header.counts.assign(header.fields.size(), 1);
120 }
121 return header.points > 0 && !header.data_type.empty();
122}

179 values[static_cast<std::size_t>(y_index)],
180 values[static_cast<std::size_t>(z_index)]});

195 }

211 header.sizes[static_cast<std::size_t>(x_index)], x) ||

213 header.sizes[static_cast<std::size_t>(y_index)], y) ||

227 std::uint32_t uncompressed_size = 0;

240 return -1;

247 return -1;
248 }
249

251 std::size_t cursor = 0;
252 for (std::size_t i = 0; i < header.fields.size(); ++i) {
253 field_block_offsets[i] = cursor;
254 cursor += static_cast<std::size_t>(header.sizes[i] * header.counts[i]) * header.points;
255 }
256 if (cursor > uncompressed.size()) {

259 return -1;
260 }

265 float z = 0.0f;
266 const unsigned char* x_ptr =

279 !detail::readFloat(y_ptr, header.sizes[static_cast<std::size_t>(y_index)], y) ||

283 return -1;
284 }

286 }

288 return static_cast<int>(points.size());
289 }
290
291 std::cerr << "Error: Unsupported PCD DATA mode '" << header.data_type << "'." << std::endl;

295inline void savePCD(const std::string& filename, const std::vector<PointXYZ>& points,
296 bool binary = false) {
297 std::ofstream file(filename, binary ? (std::ios::binary | std::ios::out) : std::ios::out);
298 if (!file.is_open()) {
299 std::cerr << "Error: Could not open file " << filename << " for writing." << std::endl;
300 return;
301 }
302
303 file << "# .PCD v.7 - Point Cloud Data file format\n";
304 file << "VERSION .7\n";
305 file << "FIELDS x y z\n";
306 file << "SIZE 4 4 4\n";
307 file << "TYPE F F F\n";
308 file << "COUNT 1 1 1\n";
309 file << "WIDTH " << points.size() << "\n";
310 file << "HEIGHT 1\n";
311 file << "VIEWPOINT 0 0 0 1 0 0 0\n";
312 file << "POINTS " << points.size() << "\n";
313 file << "DATA " << (binary ? "binary" : "ascii") << "\n";
314
315 if (binary) {
316 for (const auto& p : points) {
317 file.write(reinterpret_cast<const char*>(&p.x), sizeof(float));
318 file.write(reinterpret_cast<const char*>(&p.y), sizeof(float));
319 file.write(reinterpret_cast<const char*>(&p.z), sizeof(float));
320 }
321 } else {
322 for (const auto& p : points) {
323 file << p.x << " " << p.y << " " << p.z << "\n";
324 }
325 }
326
327 file.close();
328 std::cout << "Saved " << points.size() << " points to " << filename << std::endl;
329}
