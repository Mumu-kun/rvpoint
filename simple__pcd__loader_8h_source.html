33inline bool decompressLZF(const std::vector<unsigned char>& input,
34 std::vector<unsigned char>& output) {
35 std::size_t ip = 0;
36 std::size_t op = 0;
37
38 while (ip < input.size()) {
39 unsigned int ctrl = input[ip++];
40 if (ctrl < (1u << 5)) {
41 ctrl += 1;
42 if (op + ctrl > output.size() || ip + ctrl > input.size()) {
43 return false;
44 }
45 std::memcpy(output.data() + op, input.data() + ip, ctrl);
46 op += ctrl;
47 ip += ctrl;
48 } else {
49 unsigned int len = ctrl >> 5;
50 if (ip >= input.size()) {
51 return false;
52 }
53 std::size_t ref = op - ((ctrl & 0x1f) << 8) - 1;
54 if (len == 7) {
55 len += input[ip++];
56 if (ip >= input.size()) {
57 return false;
58 }
59 }
60 ref -= input[ip++];
61
62 if (ref >= op || op + len + 2 > output.size()) {
63 return false;
64 }
65
66 output[op++] = output[ref++];
67 output[op++] = output[ref++];
68 while (len--) {
69 output[op++] = output[ref++];
70 }
71 }
72 }
73
74 return op == output.size();
75}

77inline bool parsePCDHeader(std::istream& stream, PCDHeader& header) {
78 std::string line;
79 while (std::getline(stream, line)) {
80 if (line.empty() || line[0] == '#') {
81 continue;
82 }
83
84 std::stringstream ss(line);
85 std::string tag;
86 ss >> tag;
87 if (tag == "FIELDS") {
88 header.fields.clear();
89 std::string field;
90 while (ss >> field) {
91 header.fields.push_back(field);
92 }
93 } else if (tag == "SIZE") {
94 header.sizes.clear();
95 int value = 0;
96 while (ss >> value) {
97 header.sizes.push_back(value);
98 }
99 } else if (tag == "TYPE") {
100 header.types.clear();
101 char value = '\0';
102 while (ss >> value) {
103 header.types.push_back(value);
104 }
105 } else if (tag == "COUNT") {
106 header.counts.clear();
107 int value = 0;
108 while (ss >> value) {
109 header.counts.push_back(value);
110 }
111 } else if (tag == "POINTS") {
112 ss >> header.points;
113 } else if (tag == "WIDTH" && header.points == 0) {
114 ss >> header.points;
115 } else if (tag == "DATA") {
116 ss >> header.data_type;
117 break;
118 }
119 }
120
121 if (header.fields.empty() || header.sizes.size() != header.fields.size() ||
122 header.types.size() != header.fields.size()) {
123 return false;
124 }
125 if (header.counts.empty()) {
126 header.counts.assign(header.fields.size(), 1);
127 }
128 return header.points > 0 && !header.data_type.empty();
129}

303 bool binary = false) {
304 std::ofstream file(filename, binary ? (std::ios::binary | std::ios::out) : std::ios::out);
305 if (!file.is_open()) {

308 }
309
310 file << "# .PCD v.7 - Point Cloud Data file format\n";

315 file << "COUNT 1 1 1\n";

317 file << "HEIGHT 1\n"; // Unorganized point cloud

328 } else {
329 for (const auto& p : points) {
330 file << p.x << " " << p.y << " " << p.z << "\n";

335 std::cout << "Saved " << points.size() << " points to " << filename << std::endl;
