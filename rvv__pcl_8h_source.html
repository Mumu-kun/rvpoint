527 // Hash function: (ix, iy, iz) -> unique key

529 return (int64_t)ix + (int64_t)iy * p1_ + (int64_t)iz * p2_;
530 }
531
532 // Get cell indices for a point
533 inline void getCellIndices(float x, float y, float z, int &ix, int &iy,
534 int &iz) const {
535 ix = (int)std::floor((x - min_x_) / cell_size_);
536 iy = (int)std::floor((y - min_y_) / cell_size_);
