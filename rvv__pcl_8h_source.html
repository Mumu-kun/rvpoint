549 int64_t p1_ = 73856093LL;
550 int64_t p2_ = 19349663LL;
551
552 // Hash table: key = cell hash, value = point indices in that cell
553 std::unordered_map<int64_t, std::vector<int>> grid_;
554
555 // Bounding box
556 float min_x_, min_y_, min_z_;
557 float max_x_, max_y_, max_z_;
558 int grid_size_x_, grid_size_y_, grid_size_z_;
559
560 // Hash function: (ix, iy, iz) -> unique key
561 inline int64_t hashCell(int ix, int iy, int iz) const {
562 return (int64_t)ix + (int64_t)iy * p1_ + (int64_t)iz * p2_;
563 }
564
565 // Get cell indices for a point
566 inline void getCellIndices(float x, float y, float z, int &ix, int &iy,
567 int &iz) const {
568 ix = (int)std::floor((x - min_x_) / cell_size_);
569 iy = (int)std::floor((y - min_y_) / cell_size_);
570 iz = (int)std::floor((z - min_z_) / cell_size_);
571 }
